        }

                        sender: currentUser.name,
                        text: `Automatische Bestellung:\n${orderDetails}`,
                        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
                    }
                ]
            };

            ticketsState.unshift(newTicket);
            saveState();

            closeOrderModal();
            alert('Deine Bestellung wurde als Ticket eingereicht! Du kannst den Status im Menü unter "Bestellungen & Support" einsehen.');
            switchTab('tickets');
        }

        // --- PUBLIC CHAT ---
        function renderPublicChat() {
            const container = document.getElementById('chat-messages');
            container.innerHTML = '';

            globalChat.forEach(m => {
                const isMe = m.sender === currentUser.name;
                const msgEl = document.createElement('div');

                msgEl.innerHTML = `

                        <div class="flex items-center space-x-2 font-bold text-[10px] ${isMe ? 'text-red-100' : 'text-red-400'}">
                            <span>${m.sender}</span>
                            ${m.isAdmin ? '<span class="bg-red-950 text-red-300 text-[9px] px-1 rounded border border-red-800">ADMIN</span>' : ''}
                            <span class="text-gray-400 font-normal ml-auto">${m.time}</span>
                        </div>
                        <p class="whitespace-pre-wrap">${m.text}</p>
                    </div>
                `;
                container.appendChild(msgEl);
            });

        }

        function sendPublicMessage(e) {
            e.preventDefault();
            checkBannedStatus();

            const input = document.getElementById('chat-input');
            const text = input.value.trim();
            if (!text) return;

            globalChat.push({
                sender: currentUser.name,
                text: text,
                time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
                isAdmin: currentUser.isAdmin
            });

            saveState();
            input.value = '';
            renderPublicChat();
        }

        // --- TICKETS & ORDER STATUS TRACKING ---
        function renderTickets() {
            const list = document.getElementById('tickets-list');
            list.innerHTML = '';

            const visibleTickets = currentUser.isAdmin
                ? ticketsState
                : ticketsState.filter(t => t.userKey === currentUser.key);

            if (visibleTickets.length === 0) {
                list.innerHTML = `<div class="bg-brand-cardBg border border-brand-border p-8 text-center rounded-3xl text-gray-500 text-xs">Keine aktiven Bestellungen oder Tickets vorhanden.</div>`;
                return;
            }

            visibleTickets.forEach(t => {
                let badgeClass = 'bg-yellow-500/20 text-yellow-400 border-yellow-500/30';
                if (t.status === 'In Bearbeitung') badgeClass = 'bg-blue-500/20 text-blue-400 border-blue-500/30';
                if (t.status === 'Unterwegs / Versand') badgeClass = 'bg-purple-500/20 text-purple-400 border-purple-500/30';
                if (t.status === 'Abgeschlossen') badgeClass = 'bg-green-500/20 text-green-400 border-green-500/30';

                const card = document.createElement('div');
                card.className = "bg-brand-cardBg border border-brand-border p-4 rounded-3xl flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 hover:border-red-500/40 transition cursor-pointer";
                card.onclick = () => openTicketDetail(t.id);

                card.innerHTML = `
                    <div class="space-y-1">
                        <div class="flex items-center space-x-3">
                            <span class="font-extrabold text-white text-sm">${t.subject}</span>
                            <span class="text-[10px] px-2.5 py-0.5 rounded-full border font-bold uppercase tracking-wider ${badgeClass}">${t.status}</span>
                        </div>
                        <p class="text-xs text-gray-400">Kunde: <span class="text-red-400 font-medium">${t.userName}</span> | Nachrichten: ${t.messages.length}</p>
                    </div>
                    <button class="bg-brand-panelBg hover:bg-red-600 hover:text-white text-gray-300 text-xs font-semibold px-3.5 py-2.5 rounded-2xl border border-brand-border transition">
                        Status & Details <i class="fa-solid fa-chevron-right ml-1"></i>
                    </button>
                `;

                list.appendChild(card);
            });
        }

        function openTicketDetail(ticketId) {
            checkBannedStatus();
            currentOpenTicketId = ticketId;
            const ticket = ticketsState.find(t => t.id === ticketId);
            if (!ticket) return;

            document.getElementById('td-subject').innerText = ticket.subject;
            document.getElementById('td-user-info').innerText = `Kunde: ${ticket.userName}`;

            const badge = document.getElementById('td-status-badge');
            badge.innerText = ticket.status;
            badge.className = "px-2.5 py-1 rounded-full text-[10px] font-bold uppercase tracking-wider ";

            if (ticket.status === 'In Bearbeitung') badge.className += "bg-blue-500/20 text-blue-400 border border-blue-500/30";
            else if (ticket.status === 'Unterwegs / Versand') badge.className += "bg-purple-500/20 text-purple-400 border border-purple-500/30";
            else if (ticket.status === 'Abgeschlossen') badge.className += "bg-green-500/20 text-green-400 border border-green-500/30";
            else badge.className += "bg-yellow-500/20 text-yellow-400 border border-yellow-500/30";

            if (currentUser.isAdmin) {
                document.getElementById('td-admin-controls').classList.remove('hidden');
            } else {
                document.getElementById('td-admin-controls').classList.add('hidden');
            }

            renderTicketMessages(ticket);
            document.getElementById('ticket-detail-modal').classList.remove('hidden');
        }

        function closeTicketDetailModal() {
            document.getElementById('ticket-detail-modal').classList.add('hidden');
            currentOpenTicketId = null;
        }

        function renderTicketMessages(ticket) {
            const container = document.getElementById('td-chat-messages');
            container.innerHTML = '';

            ticket.messages.forEach(m => {
                const isMe = m.sender === currentUser.name;
                const msgEl = document.createElement('div');
                msgEl.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'}`;

                msgEl.innerHTML = `
                    <div class="max-w-[85%] ${isMe ? 'bg-red-600 text-white' : 'bg-brand-panelBg text-gray-200 border border-brand-border'} p-3 rounded-2xl text-xs space-y-1">
                        <div class="flex items-center space-x-2 font-bold text-[10px] ${isMe ? 'text-red-100' : 'text-red-400'}">
                            <span>${m.sender}</span>
                            <span class="text-gray-400 font-normal ml-auto">${m.time}</span>
                        </div>
                        <p class="whitespace-pre-wrap">${m.text}</p>
                    </div>
                `;
                container.appendChild(msgEl);
            });

            container.scrollTop = container.scrollHeight;
        }

        function sendTicketReply(e) {
            e.preventDefault();
            checkBannedStatus();
            if (!currentOpenTicketId) return;

            const input = document.getElementById('td-reply-input');
            const text = input.value.trim();
            if (!text) return;

            const ticket = ticketsState.find(t => t.id === currentOpenTicketId);
            if (ticket) {
                ticket.messages.push({
                    sender: currentUser.name,
                    text: text,
                    time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
                });

                saveState();
                input.value = '';
                renderTicketMessages(ticket);
                renderTickets();
            }
        }

        function setTicketStatus(status) {
            if (!currentUser.isAdmin || !currentOpenTicketId) return;

            const ticket = ticketsState.find(t => t.id === currentOpenTicketId);
            if (ticket) {
                ticket.status = status;
                saveState();
                openTicketDetail(currentOpenTicketId);
                renderTickets();
            }
        }

        function deleteCurrentTicket() {
            if (!currentUser.isAdmin || !currentOpenTicketId) return;
            if (!confirm('Ticket wirklich löschen?')) return;

            ticketsState = ticketsState.filter(t => t.id !== currentOpenTicketId);
            saveState();
            closeTicketDetailModal();
            renderTickets();
        }

        function openNewTicketModal() {
            document.getElementById('ticket-modal').classList.remove('hidden');
        }

        function closeTicketModal() {
            document.getElementById('ticket-modal').classList.add('hidden');
        }

        function submitNewTicket(e) {
            e.preventDefault();
            checkBannedStatus();

            const subject = document.getElementById('ticket-subject-input').value.trim();
            const msg = document.getElementById('ticket-msg-input').value.trim();

            if (!subject || !msg) return;

            const newTicket = {
                id: Date.now(),
                userKey: currentUser.key,
                userName: currentUser.name,
                subject: subject,
                status: 'Offen',
                messages: [
                    {
                        sender: currentUser.name,
                        text: msg,
                        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
                    }
                ]
            };

            ticketsState.unshift(newTicket);
            saveState();

            document.getElementById('ticket-subject-input').value = '';
            document.getElementById('ticket-msg-input').value = '';
            closeTicketModal();

            renderTickets();
        }

        // --- ADMIN PRIVATE CHAT ---
        function renderAdminChatUserList() {
            if (!currentUser || !currentUser.isAdmin) return;

            const list = document.getElementById('admin-chat-users');
            list.innerHTML = '';

            const clients = keysState.filter(k => !k.isAdmin);

            clients.forEach(c => {
                const btn = document.createElement('button');
                btn.className = `w-full text-left p-3 rounded-2xl text-xs font-semibold flex items-center justify-between transition ${activeAdminChatUserKey === c.key ? 'bg-red-600 text-white' : 'text-gray-300 hover:bg-brand-darkBg'}`;
                btn.onclick = () => selectAdminChatUser(c.key);

                btn.innerHTML = `
                    <span>${c.name}</span>
                    <i class="fa-solid fa-chevron-right text-[10px]"></i>
                `;
                list.appendChild(btn);
            });
        }

        function selectAdminChatUser(key) {
            activeAdminChatUserKey = key;
            const targetUser = keysState.find(k => k.key === key);
            if (!targetUser) return;

            document.getElementById('admin-chat-target-name').innerText = `Privatchat mit: ${targetUser.name}`;
            document.getElementById('admin-chat-input').disabled = false;
            document.getElementById('admin-chat-btn').disabled = false;

            renderAdminChatUserList();
            renderAdminPrivateMessages();
        }

        function renderAdminPrivateMessages() {
            if (!activeAdminChatUserKey) return;
            const container = document.getElementById('admin-chat-messages');
            container.innerHTML = '';

            const messages = privateChatState[activeAdminChatUserKey] || [];

            if (messages.length === 0) {
                container.innerHTML = `<p class="text-center text-gray-500 text-xs mt-10">Noch keine Nachrichten in diesem Privatchat.</p>`;
                return;
            }

            messages.forEach(m => {
                const isMe = m.sender === currentUser.name;
                const msgEl = document.createElement('div');
                msgEl.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'}`;

                msgEl.innerHTML = `
                    <div class="max-w-[80%] ${isMe ? 'bg-red-600 text-white' : 'bg-brand-panelBg text-gray-200 border border-brand-border'} p-3 rounded-2xl text-xs space-y-1">
                        <div class="flex items-center space-x-2 font-bold text-[10px] ${isMe ? 'text-red-100' : 'text-red-400'}">
                            <span>${m.sender}</span>
                            <span class="text-gray-400 font-normal ml-auto">${m.time}</span>
                        </div>
                        <p class="whitespace-pre-wrap">${m.text}</p>
                    </div>
                `;
                container.appendChild(msgEl);
            });

            container.scrollTop = container.scrollHeight;
        }

        function sendAdminPrivateMessage(e) {
            e.preventDefault();
            if (!currentUser || !currentUser.isAdmin || !activeAdminChatUserKey) return;

            const input = document.getElementById('admin-chat-input');
            const text = input.value.trim();
            if (!text) return;

            if (!privateChatState[activeAdminChatUserKey]) {
                privateChatState[activeAdminChatUserKey] = [];
            }

            privateChatState[activeAdminChatUserKey].push({
                sender: currentUser.name,
                text: text,
                time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
            });

            saveState();
            input.value = '';
            renderAdminPrivateMessages();
        }

        // --- BAN SYSTEM ---
        function renderBanList() {
            if (!currentUser || !currentUser.isAdmin) return;

            const container = document.getElementById('ban-users-list');
            container.innerHTML = '';

            const clients = keysState.filter(k => !k.isAdmin);

            if (clients.length === 0) {
                container.innerHTML = `<div class="p-6 text-center text-gray-500 text-xs">Keine registrierten Nutzer vorhanden.</div>`;
                return;
            }

            clients.forEach(c => {
                const item = document.createElement('div');
                item.className = "p-4 flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 bg-brand-cardBg hover:bg-brand-panelBg transition";

                item.innerHTML = `
                    <div>
                        <div class="flex items-center space-x-2">
                            <span class="font-bold text-white text-sm">${c.name}</span>
                            ${c.banned ? '<span class="bg-red-950 text-red-400 text-[10px] font-bold px-2 py-0.5 rounded border border-red-800">GEBANNT</span>' : '<span class="bg-green-950 text-green-400 text-[10px] font-bold px-2 py-0.5 rounded border border-green-800">AKTIV</span>'}
                        </div>
                        <p class="text-xs font-mono text-gray-400 mt-0.5">Schlüssel: ${c.key}</p>
                    </div>
                    <div>
                        ${c.banned ?
                            `<button onclick="unbanUser('${c.key}')" class="bg-gray-800 hover:bg-gray-700 text-gray-200 text-xs font-bold px-3.5 py-2.5 rounded-2xl border border-gray-700 transition">Entbannen</button>` :
                            `<button onclick="banUser('${c.key}')" class="bg-red-800 hover:bg-red-900 text-white text-xs font-bold px-3.5 py-2.5 rounded-2xl transition shadow-md">${i18n[currentLang].banBtn || 'Vom Server bannen'}</button>`
                        }
                    </div>
                `;

                container.appendChild(item);
            });
        }

        function banUser(key) {
            if (!currentUser || !currentUser.isAdmin) {
                alert('Aktion verweigert!');
                return;
            }
            const target = keysState.find(k => k.key === key);
            if (target) {
                target.banned = true;
                saveState();
                renderBanList();
            }
        }

        function unbanUser(key) {
            if (!currentUser || !currentUser.isAdmin) {
                alert('Aktion verweigert!');
                return;
            }
            const target = keysState.find(k => k.key === key);
            if (target) {
                target.banned = false;
                saveState();
                renderBanList();
            }
        }

        // --- KEY GENERATOR ---
        function generateNewKey(e) {
            e.preventDefault();
            if (!currentUser || !currentUser.isAdmin) {
                alert('Aktion verweigert!');
                return;
            }

            const nameInput = document.getElementById('gen-key-name');
            const name = nameInput.value.trim();
            if (!name) return;

            const randomCode = Math.random().toString(36).substring(2, 6).toUpperCase() + '-' + Math.random().toString(36).substring(2, 6).toUpperCase();
            const newKey = `VIP-${randomCode}`;

            keysState.push({
                key: newKey,
                name: name,
                isAdmin: false,
                banned: false
            });

            saveState();
            nameInput.value = '';
            renderKeysList();
        }

        function renderKeysList() {
            if (!currentUser || !currentUser.isAdmin) return;

            const container = document.getElementById('keys-list');
            container.innerHTML = '';

            keysState.forEach(k => {
                const item = document.createElement('div');
                item.className = "bg-brand-panelBg border border-brand-border p-3.5 rounded-2xl flex items-center justify-between";

                item.innerHTML = `
                    <div>
                        <span class="font-bold text-xs text-white">${k.name}</span>
                        ${k.isAdmin ? '<span class="ml-2 bg-red-900 text-red-200 text-[9px] px-1.5 py-0.5 rounded font-bold">ADMIN</span>' : ''}
                        <p class="text-xs font-mono text-red-400 mt-0.5">${k.key}</p>
                    </div>
                    <button onclick="copyToClipboard('${k.key}')" class="text-gray-400 hover:text-red-400 p-2 text-xs" title="Schlüssel kopieren">
                        <i class="fa-solid fa-copy text-sm"></i>
                    </button>
                `;

                container.appendChild(item);
            });
        }

        function copyToClipboard(text) {
            const tempInput = document.createElement('input');
            tempInput.value = text;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand('copy');
            document.body.removeChild(tempInput);
            alert(`Schlüssel "${text}" in die Zwischenablage kopiert!`);
        }

        // --- MULTI-LANGUAGE TRANSLATOR ---
        function changeLanguage(lang) {
            currentLang = lang;
            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (i18n[lang] && i18n[lang][key]) {
                    el.innerText = i18n[lang][key];
                }
            });
        }

            // Reset state to ensure valid initial data
            localStorage.setItem('vip_keys', JSON.stringify(keysState));
            saveState();
            document.getElementById('products-grid').innerHTML = '';
        };
